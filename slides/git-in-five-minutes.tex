\begin{frame}[t]{How Git helps}
\smalltext{A \textbf{repository} is your project together with its Git history.}\par
\vspace{11mm}
\begin{columns}[T,onlytextwidth]
\begin{column}{.29\textwidth}
\labeltext{1 / EDIT}\par\vspace{4mm}
{\fontsize{26}{32}\selectfont Your files}\par\vspace{5mm}
\smalltext{Save your changes in the editor.}
\end{column}
\begin{column}{.29\textwidth}
\labeltext{2 / SELECT}\par\vspace{4mm}
{\fontsize{26}{32}\selectfont Staging area}\par\vspace{5mm}
\smalltext{\texttt{git add}\par Select current content for the next commit.}
\end{column}
\begin{column}{.29\textwidth}
\labeltext{3 / RECORD}\par\vspace{4mm}
{\fontsize{26}{32}\selectfont Commit}\par\vspace{5mm}
\smalltext{\texttt{git commit}\par Record a snapshot with a message.}
\end{column}
\end{columns}
\vspace{10mm}
\textcolor{Teal}{Saving a file does not create a commit.}\par
\vspace{3mm}
\smalltext{Edit after \texttt{git add}? Add again to include the later edits.}\par
\vfill
{\color{Muted}\fontsize{16}{21}\selectfont Commits stay on your computer. An external backup is separate.}\par
\vspace{6mm}
\end{frame}
